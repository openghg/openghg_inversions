Customising the complete RHIME workflow#

The copied runner below is an advanced, version-coupled escape hatch for scientific changes that do not fit an ordinary run_rhime option. It makes the major stages visible while importing their implementations from the supported public RHIME API. Prefer the standard runner when configuration is enough, and prefer run_rhime_from_prepared_inputs when replaying prepared inputs or deliberately starting from a different preparation graph.

The deliberate change is the likelihood passed to build_standard_rhime_model: the example selects build_absolute_sigma_gaussian_likelihood in place of RHIME’s default pollution-event-scaled Gaussian. Acquisition, filtering, basis construction, labelled input assembly, the eager PyMC boundary, sampling, predictive selection, filenames, and output handling remain library-owned.

In a project created with the OpenGHG project cookiecutter, copy this exact module to src/<package_name>/rhime_runner.py. Keep the short orchestration spine in the project and import the scientific stage implementations from openghg_inversions.rhime; do not copy those implementations into the project.

Run it with a normal RHIME configuration and optional overrides:

python -m package_name.rhime_runner config.ini \
    --start-date 2020-01-01 --end-date 2020-02-01 \
    --output-path outputs --draws 1000 --tune 1000 --chains 4

Less common Python/config options can be supplied as a JSON object:

python -m package_name.rhime_runner config.ini \
    --kwargs '{"reload_merged_data": true, "output_format": "inv_out"}'

The source below is also imported and executed by the integration test, so the documentation and runnable example cannot drift apart.

  1"""Advanced copy-and-modify runner for a standard RHIME inversion.
  2
  3This example deliberately replaces RHIME's default likelihood with the public
  4absolute-sigma Gaussian helper.  The orchestration is intentionally copied so
  5that a project can make deeper scientific changes while continuing to reuse
  6the supported acquisition, preparation, model, sampling, and output stages.
  7Use :func:`run_custom_rhime` from Python or :func:`main` from the command line.
  8The standard single-sector stages are preserved, model inputs materialize only
  9at the explicit PyMC boundary, and a run may acquire or reload data, sample a
 10model, and write its configured outputs.
 11"""
 12
 13from __future__ import annotations
 14
 15import argparse
 16from collections.abc import Sequence
 17import json
 18from pathlib import Path
 19from time import perf_counter
 20from typing import Any
 21
 22from openghg_inversions.rhime import (
 23    RhimeResult,
 24    assemble_rhime_inputs,
 25    build_absolute_sigma_gaussian_likelihood,
 26    build_rhime_basis,
 27    build_rhime_sensitivities,
 28    build_standard_rhime_model,
 29    filter_rhime_observations,
 30    make_standard_rhime_result,
 31    materialize_pymc_inputs,
 32    params_from_config,
 33    resolve_rhime_options,
 34    retrieve_or_reload_rhime_data,
 35    sample_rhime_model,
 36    with_prepared_rhime_sites,
 37)
 38
 39
 40def run_custom_rhime(
 41    *,
 42    config_file: str | Path | None = None,
 43    **kwargs: Any,
 44) -> RhimeResult:
 45    """Run standard RHIME with an absolute-sigma Gaussian likelihood.
 46
 47    Args:
 48        config_file: Optional RHIME INI configuration file.
 49        **kwargs: Standard RHIME option names that override values from
 50            ``config_file``.
 51
 52    Returns:
 53        The sampled result and any outputs requested by the RHIME options.
 54
 55    Notes:
 56        This workflow may retrieve or reload data, eagerly materializes PyMC
 57        model inputs, runs sampling, and writes outputs requested by the
 58        resolved RHIME options.
 59    """
 60    params = (
 61        params_from_config(config_file, extra_kwargs=kwargs, normalise=False)
 62        if config_file is not None
 63        else dict(kwargs)
 64    )
 65    setup = resolve_rhime_options(params=params, multisector=False)
 66
 67    merged = retrieve_or_reload_rhime_data(setup.data_args, multisector=False)
 68    filtered = filter_rhime_observations(merged, setup.data_args)
 69    basis_functions = build_rhime_basis(filtered, setup.data_args)
 70    site_data = build_rhime_sensitivities(
 71        filtered,
 72        basis_functions,
 73        setup.data_args,
 74        multisector=False,
 75    )
 76    prepared = assemble_rhime_inputs(filtered, basis_functions, site_data, setup.data_args)
 77    run_spec = with_prepared_rhime_sites(setup.run_spec, prepared)
 78
 79    model_inputs = materialize_pymc_inputs(
 80        prepared,
 81        aggregation_error_mode=run_spec.model.aggregation_error_mode,
 82    )
 83    build_and_sample_start = perf_counter()
 84    model_build_result = build_standard_rhime_model(
 85        prepared=prepared,
 86        model_inputs=model_inputs,
 87        run_spec=run_spec,
 88        # This is the deliberate scientific replacement in the copied runner.
 89        likelihood_builder=build_absolute_sigma_gaussian_likelihood,
 90    )
 91    idata = sample_rhime_model(
 92        model_build_result,
 93        setup.sampler,
 94        use_variable_roles=True,
 95    )
 96
 97    return make_standard_rhime_result(
 98        prepared=prepared,
 99        run_spec=run_spec,
100        sampler=setup.sampler,
101        model_build_result=model_build_result,
102        idata=idata,
103        build_and_sample_seconds=perf_counter() - build_and_sample_start,
104        likelihood_builder=build_absolute_sigma_gaussian_likelihood,
105    )
106

108def _json_object(value: str) -> dict[str, Any]:
109    """Parse one command-line JSON object containing additional RHIME options.
110
111    Args:
112        value: JSON text to parse.
113
114    Returns:
115        The decoded JSON object.
116
117    Raises:
118        argparse.ArgumentTypeError: If the text is invalid JSON or decodes to
119            a non-object value.
120    """
121    try:
122        parsed = json.loads(value)
123    except json.JSONDecodeError as exc:
124        raise argparse.ArgumentTypeError(f"invalid JSON: {exc.msg}") from exc
125    if not isinstance(parsed, dict):
126        raise argparse.ArgumentTypeError("--kwargs must decode to a JSON object")
127    return parsed
128
129
130def main(argv: Sequence[str] | None = None) -> RhimeResult:
131    """Parse command-line options and run the customized RHIME workflow.
132
133    This performs the same acquisition, materialization, sampling, and output
134    side effects as :func:`run_custom_rhime`.
135
136    Args:
137        argv: Command-line tokens excluding the program name. ``None`` reads
138            tokens from :data:`sys.argv`.
139
140    Returns:
141        The RHIME result returned by :func:`run_custom_rhime`.
142
143    Raises:
144        SystemExit: If argument parsing fails or help is requested.
145    """
146    parser = argparse.ArgumentParser(description=__doc__)
147    parser.add_argument("config_file", nargs="?", type=Path, help="RHIME INI configuration file")
148    parser.add_argument("--start-date")
149    parser.add_argument("--end-date")
150    parser.add_argument("--output-path", type=Path)
151    parser.add_argument("--output-name")
152    parser.add_argument("--draws", type=int)
153    parser.add_argument("--tune", type=int)
154    parser.add_argument("--chains", type=int)
155    parser.add_argument(
156        "--kwargs",
157        type=_json_object,
158        default={},
159        metavar="JSON",
160        help="additional RHIME options as a JSON object",
161    )
162    args = parser.parse_args(argv)
163
164    overrides = dict(args.kwargs)
165    for name in ("start_date", "end_date", "output_path", "output_name", "draws", "tune", "chains"):
166        value = getattr(args, name)
167        if value is not None:
168            overrides[name] = value
169    return run_custom_rhime(config_file=args.config_file, **overrides)
170
171
172if __name__ == "__main__":
173    main()
